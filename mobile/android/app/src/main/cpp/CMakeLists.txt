cmake_minimum_required(VERSION 3.22.1)
project(diagrammer_mnn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(diagrammer_mnn SHARED mnn_bridge.cpp)
if(MNN_ROOT)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "The MNN mobile build targets arm64-v8a only.")
    endif()
    if(NOT EXISTS "${MNN_ROOT}/transformers/llm/engine/include/llm/llm.hpp")
        message(FATAL_ERROR "MNN_ROOT must be the MNN 3.6.1 source directory.")
    endif()
    # Skip upstream post-build copying on combined Android OBJECT targets.
    set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
    set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
    set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
    set(MNN_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
    set(MNN_BUILD_OPENCV ON CACHE BOOL "" FORCE)
    set(MNN_IMGCODECS ON CACHE BOOL "" FORCE)
    set(MNN_SUPPORT_TRANSFORMER_FUSE ON CACHE BOOL "" FORCE)
    set(MNN_LLM_BUILD_DEMO OFF CACHE BOOL "" FORCE)
    set(MNN_BUILD_TEST OFF CACHE BOOL "" FORCE)
    set(MNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
    set(MNN_USE_SSE OFF CACHE BOOL "" FORCE)
    set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
    add_subdirectory("${MNN_ROOT}" "${CMAKE_BINARY_DIR}/mnn")
    target_include_directories(diagrammer_mnn PRIVATE "${MNN_ROOT}/include" "${MNN_ROOT}/transformers/llm/engine/include")
    target_link_libraries(diagrammer_mnn MNN)
    target_compile_definitions(diagrammer_mnn PRIVATE DIAGRAMMER_WITH_MNN=1)
endif()
